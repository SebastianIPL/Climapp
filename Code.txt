import { Component, OnInit, OnDestroy, ChangeDetectorRef } from '@angular/core';
import { CommonModule } from '@angular/common';
import { RouterModule } from '@angular/router';

interface ForecastPeriod {
  timeLabel: string;
  aqi: number;
  status: string;
}

interface AirData {
  city: string;
  station: string;
  aqi: number;
  statusClass: string;
  statusText: string;
  canGoOut: boolean;
  verdictTitle: string;
  verdictMessage: string;
  pm25: number | string;
  pm10: number | string;
  lastUpdated: string;
  timeline: ForecastPeriod[];
  tips: { title: string; desc: string }[];
}

@Component({
  selector: 'app-home',
  templateUrl: 'home.page.html',
  styleUrls: ['home.page.scss'],
  standalone: true,
  imports: [CommonModule, RouterModule]
})
export class HomePage implements OnInit, OnDestroy {
  isLoading: boolean = false;
  private refreshTimer: any;

  airData: AirData = {
    city: 'Maipú',
    station: 'Estación Cerrillos / Poniente',
    aqi: 50,
    statusClass: 'status-good',
    statusText: 'Bueno',
    canGoOut: true,
    verdictTitle: 'Seguro para salir',
    verdictMessage: 'Condiciones favorables en Maipú.',
    pm25: '--',
    pm10: '--',
    lastUpdated: 'Conectando...',
    timeline: [],
    tips: [
      { title: 'Deporte al aire libre', desc: 'Condiciones aceptables para la práctica deportiva.' },
      { title: 'Ventilación', desc: 'Buen momento para ventilar dependencias del hogar.' },
      { title: 'Tránsito local', desc: 'Atención a ejes con alta carga vehicular en Maipú.' }
    ]
  };

  constructor(private cdr: ChangeDetectorRef) {}

  ngOnInit() {
    this.loadMaipuDataRealTime();

    // Actualización automática recurrente cada 15 minutos
    this.refreshTimer = setInterval(() => {
      this.loadMaipuDataRealTime();
    }, 15 * 60 * 1000);
  }

  ngOnDestroy() {
    if (this.refreshTimer) {
      clearInterval(this.refreshTimer);
    }
  }

  async loadMaipuDataRealTime() {
    this.isLoading = true;
    this.cdr.detectChanges();

    try {
      // Estación física oficial Red SINCA Cerrillos/Poniente con tu token activo
      const url = 'https://api.waqi.info/feed/@262/?token=f8de41860dad59cc72b536922daede3f03088caf';
      const response = await fetch(url);
      const result = await response.json();

      if (result.status === 'ok' && result.data) {
        const data = result.data;
        
        // AQI oficial medido en la estación Cerrillos/Poniente
        const currentAqi = Number(data.aqi ?? 50);
        const rawPm25 = data.iaqi?.pm25?.v ?? '--';
        const rawPm10 = data.iaqi?.pm10?.v ?? '--';

        const pm25 = typeof rawPm25 === 'number' ? rawPm25.toFixed(1) : String(rawPm25);
        const pm10 = typeof rawPm10 === 'number' ? rawPm10.toFixed(1) : String(rawPm10);

        const timeline: ForecastPeriod[] = [
          { timeLabel: 'Mañana (08:00)', aqi: Math.max(20, currentAqi - 5), status: this.getStatusLabel(Math.max(20, currentAqi - 5)) },
          { timeLabel: 'Mediodía (13:00)', aqi: currentAqi, status: this.getStatusLabel(currentAqi) },
          { timeLabel: 'Tarde (18:00)', aqi: currentAqi + 5, status: this.getStatusLabel(currentAqi + 5) },
          { timeLabel: 'Noche (22:00)', aqi: currentAqi + 10, status: this.getStatusLabel(currentAqi + 10) }
        ];

        this.airData = this.buildAirVerdict(currentAqi, pm25, pm10, timeline);
      }
    } catch (err) {
      console.warn('Error al conectar con la estación de Cerrillos/Maipú:', err);
    } finally {
      this.isLoading = false;
      this.cdr.detectChanges();
    }
  }

  private getStatusLabel(aqi: number): string {
    if (aqi <= 50) return 'Bueno';
    if (aqi <= 100) return 'Moderado';
    if (aqi <= 150) return 'Dañino p/ sensibles';
    if (aqi <= 200) return 'Dañino';
    return 'Muy Dañino';
  }

  private buildAirVerdict(aqi: number, pm25: string, pm10: string, timeline: ForecastPeriod[]): AirData {
    let statusClass = 'status-good';
    let statusText = 'Bueno';
    let canGoOut = true;
    let verdictTitle = 'Seguro para salir';
    let verdictMessage = 'Condiciones favorables en Maipú.';

    if (aqi > 50 && aqi <= 100) {
      statusClass = 'status-moderate';
      statusText = 'Moderado';
      canGoOut = true;
      verdictTitle = 'Moderado: Salida permitida';
      verdictMessage = 'La calidad del aire es aceptable. Riesgo menor para personas ultrasensibles.';
    } else if (aqi > 100 && aqi <= 150) {
      statusClass = 'status-sensitive';
      statusText = 'Dañino p/ sensibles';
      canGoOut = false;
      verdictTitle = 'Precaución en exteriores';
      verdictMessage = 'Grupos de riesgo deben moderar actividades físicas intensas.';
    } else if (aqi > 150) {
      statusClass = 'status-danger';
      statusText = 'Dañino';
      canGoOut = false;
      verdictTitle = 'No recomendado salir';
      verdictMessage = 'Niveles elevados de contaminación. Suspender deporte exterior.';
    }

    return {
      city: 'Maipú',
      station: 'Estación Cerrillos / Red SINCA',
      aqi,
      statusClass,
      statusText,
      canGoOut,
      verdictTitle,
      verdictMessage,
      pm25,
      pm10,
      lastUpdated: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
      timeline,
      tips: this.airData.tips
    };
  }
}
